                    ; 64 health lines on screen
                    ; 16 tiles across
                    ; 256 possible values
                    ; displaying 1/4th the range
      
d0d4  a0 00         ldy #$00

d0d8  85 86         sta temp00 ; life value floor for tile

                .find_top_tile
