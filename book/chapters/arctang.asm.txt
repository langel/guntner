                .bottom_quadrant

                .top_quadrant

                .right_quadrant

                .left_quadrant

                .x_quadrant_done

ce43  aa            tax
                    ; quadrant is now in temp00
                    ; start finding that region

                .x_less_than_y

                .x_greater_or_equal_y
