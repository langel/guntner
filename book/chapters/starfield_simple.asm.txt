                starfield_init: subroutine
                    ; set starfield state
d4ff  a9 00         lda #$00
d501  85 31         sta starfield_state
                    ; set bg tile palette attributes / colors
                    ; $23c0 and $27c0
                    ; page 1 attributes
                    PPU_SETADDR $23c0
d503  a9 23         lda #>$23c0
d505  8d 06 20      sta PPU_ADDR
d508  a9 c0         lda #<$23c0
d50a  8d 06 20      sta PPU_ADDR
d50d  a9 14         lda #%00010100
d50f  a2 d8         ldx #$d8
                .23c0_loop
d511  8d 07 20      sta PPU_DATA
d514  e8            inx
d515  d0 fa         bne .23c0_loop
                    ; COLOR SPLIT
                    ; set split color attr for top row
                    PPU_SETADDR $23e8
d517  a9 23         lda #>$23e8
d519  8d 06 20      sta PPU_ADDR
d51c  a9 e8         lda #<$23e8
d51e  8d 06 20      sta PPU_ADDR
d521  a9 f1         lda #%11110001
d523  a2 f8         ldx #$f8
                .27e8_loop
d525  8d 07 20      sta PPU_DATA
d528  e8            inx
d529  d0 fa         bne .27e8_loop
                    ; page 2 attributes
                    PPU_SETADDR $27c0
d52b  a9 27         lda #>$27c0
d52d  8d 06 20      sta PPU_ADDR
d530  a9 c0         lda #<$27c0
d532  8d 06 20      sta PPU_ADDR
d535  a9 14         lda #%00010100
d537  a2 d8         ldx #$d8
                .27c0_loop
d539  8d 07 20      sta PPU_DATA
d53c  e8            inx
d53d  d0 fa         bne .27c0_loop
                    ; make sure sprite0 has something to collide with
d53f  20 dd d5      jsr starfield_draw_dash_top_bar_nametable0
                    ; draw stars on nametable 1
d542  a9 04         lda #CTRL_INC_32
d544  8d 00 20      sta PPU_CTRL
                    ;stx starfield_page
d547  a2 00         ldx #$00
                .draw_stars_loop
d549  86 32         stx starfield_column
d54b  20 52 c1      jsr get_next_random
d54e  20 9c d5      jsr starfield_cache_next_col
d551  a9 20         lda #$20
d553  8d 06 20      sta PPU_ADDR
d556  a5 32         lda starfield_column
d558  8d 06 20      sta PPU_ADDR
d55b  a0 15         ldy #$15
                .transfer_loop
d55d  b9 d0 00      lda star_cache,y
d560  8d 07 20      sta PPU_DATA
d563  88            dey
d564  d0 f7         bne .transfer_loop
d566  e8            inx
d567  e0 20         cpx #$20
d569  d0 de         bne .draw_stars_loop

d56d  8d 00 20      sta PPU_CTRL
d570  60            rts
