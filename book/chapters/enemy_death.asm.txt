                clear_all_enemies: subroutine
                    ; clears all enemy ram
dd32  a2 f0         ldx #$f0
dd34  a9 00         lda #$00
                .enemy_loop
dd36  ca            dex
dd37  9d 00 03      sta enemy_ram_type,x
dd3a  d0 fa         bne .enemy_loop
                    ; clears all enemy sprites
dd3c  a9 ff         lda #$ff

                enemy_clear: subroutine
                    ; clear OAM data
dd47  a4 77         ldy enemy_oam_offset
dd49  a9 ff         lda #$ff
dd4b  a2 03         ldx #$03
                .enemy_oam_loop
dd4d  99 00 02      sta oam_ram_y,y
dd50  c8            iny
dd51  ca            dex
dd52  d0 f9         bne .enemy_oam_loop
                    ; clear enemy data
dd54  a6 76         ldx enemy_ram_offset

                .enemy_ram_loop

dd5e  88            dey
dd5f  d0 f9         bne .enemy_ram_loop
                    ; enemy_ram_ex is allowed to carry over
                    ; entities using _ex should set it on spawn
dd61  60            rts
      
      
      
                enemy_death_handler: subroutine
                    ; requires
                    ; enemy_ram_offset
                    ; enemy_oam_offset
      
                    ; don't handle a non entity

dd64  bd 00 03      lda enemy_ram_type,x
dd67  f0 59         beq .done

                    ; track deaths of phase spawns

                    ; if less than 0 set to 0

                .not_phase_spawn

                    ; 4 sprite cleanup

                    ; move other sprites off screen

dd9f  99 0c 02      sta oam_ram_y+12,y
dda2  d0 09         bne .check_for_powerup_drop
                .dont_cleanup_4_sprite
      
                    ; 2 sprite cleanup

                .check_for_powerup_drop

                    ; change it into crossbones!

                .done
ddc2  60            rts
