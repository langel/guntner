                timer_update: subroutine
                    ; UPDATE TIMER DIGITS
                    ; but not if game is over
d644  a5 63         lda phase_end_game

                    ; frames 1s

d64a  a9 6a         lda #char_set_offset+$a

                    ; frames 10s

d666  d0 2c         bne .timer_done

                    ; seconds 10s
d66c  e6 47         inc timer_seconds_10s

d672  d0 20         bne .timer_done

d67e  d0 14         bne .timer_done

                    ; we really checking this?
                    ; let it go into tile limbo
                    ; GAME OVER at 99:59.59 !!
