                skully_cycle: subroutine
eeea  a9 10         lda #$10
eeec  85 3d         sta collision_0_w
eeee  85 3e         sta collision_0_h
eef0  20 eb db      jsr enemy_handle_damage_and_death
      
eef3  a6 76         ldx enemy_ram_offset
eef5  a4 77         ldy enemy_oam_offset
eef7  a9 01         lda #$01 ; set mirror flag
eef9  9d 07 03      sta enemy_ram_ex,x
                    ; let's find what frame we're on
eefc  bd 05 03      lda enemy_ram_ac,x
eeff  4a            lsr
ef00  4a            lsr
ef01  4a            lsr
ef02  4a            lsr
ef03  4a            lsr
ef04  0a            asl
                    ; accumulator is now in 0..7 range except x2
                    ; sprite
ef05  20 b7 da      jsr sprite_4_set_sprite
                    ; x pos
ef08  bd 02 03      lda enemy_ram_x,x
ef0b  20 df da      jsr sprite_4_set_x
                    ; y pos
ef0e  bd 03 03      lda enemy_ram_y,x
ef11  20 ef da      jsr sprite_4_set_y
                    ; update spinning counter
ef14  a9 07         lda #$07
ef16  18            clc
ef17  7d 05 03      adc enemy_ram_ac,x
ef1a  9d 05 03      sta enemy_ram_ac,x
                    ; move skully
ef1d  20 28 ef      jsr skully_handle_movement
                    ; palette
ef20  a9 03         lda #$03
ef22  20 9d da      jsr sprite_4_set_palette
                .skully_done
ef25  4c 92 db      jmp update_enemies_handler_next
      
      
                skully_handle_movement: subroutine
