                player_gun_fire: subroutine
                    ; temp00 holds five_directions == true

                .bullet_loop
f372  b9 00 02      lda oam_ram_y,y
                    ; check if bullet is spawned

                    ; trigger pewpew

f381  a5 15         lda player_y_hi
f383  18            clc

                    ; direction
f389  a5 86         lda temp00
